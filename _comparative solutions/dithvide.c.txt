void conv16(unsigned char *p, int r, int g, int b)
{
	*(unsigned short *)p=((r>>3)<<11) + ((g>>2)<<5) + (b>>3);
}

{

void __inline__ purecols(int a, int b, int c, int d, int e, int f)
	{
	memcpy(pure,pall+a+e,sizeof(struct rgbpixel));
	memcpy(pure+1,pall+b+e,sizeof(struct rgbpixel));
	memcpy(pure+2,pall+c+f,sizeof(struct rgbpixel));
	memcpy(pure+3,pall+d+f,sizeof(struct rgbpixel));
	for(f=0;f<12;f++)
	  ((int *)pure)[f] = tvg[((int *)pure)[f]];
	}

void __inline__ greyscale(struct rgbpixel *p, int i)
    {
    while(i--)
	{
	p->r=p->g=p->b=(RG*p->r+GG*p->g+BG*p->b)/SH;
	p++;
	}
    }

void __inline__ basecols(int g)
	  {
	  int i, j, k=0;
	  for(i=0;i<2;i++)
	    for(j=2;j<4;j++,k++)
	      {
	      base[k].r=(pure[i].r+pure[j].r)>>1;
	       base[k].g=(pure[i].g+pure[j].g)>>1;
	        base[k].b=(pure[i].b+pure[j].b)>>1;
	      }
	  if(g)
	    greyscale(base,4);
	  }

	long __inline__ ratepair(void)

					if(pick<dist)

						p=k;
						dist=pick;
					}
				}
				rg-=pure[p].g;
			      }

				  r=pure[k].r-rr;
				  g=pure[k].g-rg;
				  b=pure[k].b-rb;
					pick=r*r+g*g+b*b;;
					if(pick<dist)
					{

						dist=pick;
					}

//				if(!dither)

			      }
				
				offset = (offset + ((3*ran)/5)) % ran;
//				printf("X%d %dX",offset,dist);
				return dist+s;
	}

//	      d=d+(flatten*O_SAT)/100;
//	      if(c<0) c=0;
//	      if(d>O_SAT) d=O_SAT;
//	      }
	  ordercols(h);
	  ordercols(h+3);
	  
	  diff=MAXDIST;
	  for(f=(brr?pls+1:0);f>=0;f-=pls+1)
	    {

	    j=0;

	for(c=0;c<pls;c++)
	for(b=pls;b>a;b--)
	for(d=pls;d>c;d--)

				suma+=ratecols();
				if(suma>=diff) break;
				rtmp+=1-2*(i&1);
			}		
			if(suma>=diff) break;
		}

		for(j=PA_X*x;j++<PA_X*x+PA_X;)
		{
			rtmp=rgb[ay*y+(ay-1)*(j&1)]+j;

		}		  
		if(suma<diff)
		{
			diff=suma;
			savecols(a,b,c,d,e,f);
		}
	}
		if(e&f) break;
		if(f) e=f;
		f=pls+1;
	}while(brr);
}

void rgbadd(unsigned long *d, int x, int y, int m, int n)
{
	int c;
	if(m>=0)
	{
		m=wlong-m;
		if(m>vrx) m=vrx;
	}
	else
	{
		m+=vrx;
		if(m<0)
			m=0;
	}
	if(n>=0)
	{
		n=wlong-n;
		if(n>vrx) n=vrx;
	}
	else
	{
		n+=vrx;
		if(n<0)
			n=0;
	}    
	c=m*n;
	rgb[y][x].r+=c*((unsigned char *)d)[2]/(vrx*vrx);
	rgb[y][x].g+=c*((unsigned char *)d)[1]/(vrx*vrx);
	rgb[y][x].b+=c*((unsigned char *)d)[0]/(vrx*vrx);
}

void xorme(unsigned char *t)
{
	int c=bpp;
	while(c--)
		t[c]^=XORVAL;
}

void wtoggle(void)
{
	int i, h=jdecs.output_width-wx, v=vry*wlong/vrx-2;
	unsigned char *t=bram+bpp*(jdecs.output_width*wy+wx), *d;

	if(h>wlong)
		h=wlong;
	d=t;
	for(i=h;i--;d+=bpp)
	xorme(d);
	d=t+bpp*jdecs.output_width;
	i=wy+1;
	while(v-- && i<jdecs.output_height)
	{
		xorme(d);
		if(h==wlong)
			xorme(d+bpp*(h-1));
		d+=bpp*jdecs.output_width;
		i++;
	}
	if(i<jdecs.output_height)
		for(i=h;i--;d+=bpp)
	xorme(d);
}

				  }
				ptmp->g=l;
				vtmp+=bpp;
				ptmp++;
			}
			if(!rgbtest) killwindow();
			free(data);
			printf("Selection: %d..%d x %d..%d\nScaling: %d%% to %d x %d\n",wx,wx+wlong-1,wy,wy-1+(wlong*vry)/vrx,100*vrx/wlong,vrx,vry);
			sprintf(u,"Scaled to %d x %d",vrx,vry);
			popwindow(vram,vrx,vry,u);
			XPutImage(dis,win,gc,xim,0,0,0,0,vrx,vry);
			XSelectInput(dis,win,ExposureMask|KeyPressMask);
			XFlush(dis);
			n=1; m=1;
			while(n)
			{
				XNextEvent(dis,&xev);
				switch(xev.type)
				{
					case Expose:
						XPutImage(dis,win,gc,xim,0,0,0,0,vrx,vry);
						XFlush(dis);
						break;
					case KeyPress:
						if(xev.xkey.keycode==ENDKEY && !m)
							n=0;
						if(xev.xkey.keycode==ORIGKEY && !m)
						{
						if(vrams == 1)
							switch(l)
							  {
							  case 0:
							  case 1: ban("Color picture");
								  l=1;
								  break;
							  case 2:
							  case 3: ban("Picture bitmap");
								  l=3;
								  break;
							  default: ban("Greyscale picture");
								   l=6;
							  }
						else
							switch(l)
							  {
							  case 0: ban("Even VRAM"); break;
							  case 1: ban("Odd VRAM"); break;
							  case 2: ban("Even bitmap"); break;
							  case 3: ban("Odd bitmap"); break;
							  case 4: ban("Bitmap contents"); break;
							  case 5: ban("Average picture"); break;
							  default: ban("Greyscale picture");
							  }
							vtmp=vram;
							for(i=0;i<vry;i++)
							for(j=0;j<vrx/PA_X;j++)
							{
								    decodeatr(i/ay,j);
								    if(vrams == 1)
								      {
								      memcpy(base,pure,sizeof(base));
								      greyscale(base,4);
								      }
								    else
								      basecols(l == 6);
								    r=atr[i/ay][j].b[0][i%ay];
								    s=atr[i/ay][j].b[1][i%ay];
								    o=(l&1)?s:r;
								  
							      for(k=8;k--;)
								{
								  switch(l)
								    {
								    case 0:
								    case 1: r=(1&(o>>7))+(2&(l<<1));
								      conv(vtmp,tvd[pure[r].r],tvd[pure[r].g],tvd[pure[r].b]); break;
								    case 2:
								    case 3: r=(o&128)?0:MAXVAL; conv(vtmp,r,r,r); break;
								    case 4: o=((MAXVAL*(3-((2&(r>>6))+(1&(s>>7)))))/3); conv(vtmp,tvd[o],tvd[o],tvd[o]); break;
								    default: o=(2&(r>>6))+(1&(s>>7)); conv(vtmp,tvd[base[o].r],tvd[base[o].g],tvd[base[o].b]);
								    }
								  r<<=1;
								  s<<=1; 
								  o<<=1;
								  vtmp+=bpp;
								}
							}
							XPutImage(dis,win,gc,xim,0,0,0,0,vrx,vry);
							XFlush(dis);
							l++;
							if(l>6) l=0;
						}	
						if(xev.xkey.keycode==ORIGKEY && m)
						{
							ban("Color matching...");
							for(i=vry/ay;i--;)
							{
							  XFlush(dis);
								for(j=vrx/PA_X;j--;)
								cmpatr(i,j);
								vtmp=vram+bpp*ay*vrx*i;
								for(p=ay;p--;)
								  for(l=0;l<vrx/PA_X;l++)
								    {
								      memcpy(base,atr[i][l].base,sizeof(base));
								for(o=0;o<PA_X;vtmp+=bpp,o++)
								  if(vrams == 1)
								    conv(vtmp,tvd[base[o>>2].r],tvd[base[o>>2].g],tvd[base[o>>2].b]);
								  else
								    conv(vtmp,tvd[base[o>>1].r],tvd[base[o>>1].g],tvd[base[o>>1].b]);
								    }							
								XPutImage(dis,win,gc,xim,0,0,0,0,vrx,vry);
								
							}
							//getchar();
							printf("Mapped, ");
							ban("Dithering...");
							XFlush(dis);
							fflush(stdout);
							vtmp=vram; ptmp=(struct rgbpixel *)rgb;
							if(vrams == 1 && !grey )
							  greyscale((struct rgbpixel *)rgb,sizeof(rgb)/sizeof(struct rgbpixel));
							for(i=0;i<vry;i++)
							{
							struct rgbpixel prevp={0,0,0};
							
								if((i&1) && (!dither))
									ptmp--;
								else
									ptmp++;
								for(j=0;j<vrx;j++)
								{
									if(!(j%PA_X))
										{
										if((i&1) && (!dither))
										memcpy(base,atr[i/ay][(vrx-1-j)/PA_X].base,sizeof(base));
										    else
										memcpy(base,atr[i/ay][j/PA_X].base,sizeof(base));
										if(!grey && (vrams == 1))
										  greyscale(base,2);
										  
										if(dither)
										    {
										    if(vrams == 1 || grey)
											{
											m=O_SAT; p=0;
											for(k=((vrams == 1)?2:4);k--;)
											    {
											    //////////uprava
											    l=(RG*base[k].r + GG*base[k].g + BG*base[k].b)/SH;
											    //l=tvd[(RG*base[k].r + GG*base[k].g + BG*base[k].b)/SH];
											    if(l<m) m=l;
											    if(l>p) p=l;
											    }
											r=p-m;
											}
										    else
											r=O_AMP;
										    }
										}
									if(dither)
									    {
									    m=matrixes[dither-1].w;
									    p=matrixes[dither-1].h;
									    l=((*(matrixes[dither-1].m+m*(i%p) + (j%m)) - ((p*m)>>1))*r)/(p*m);
									    ptmp->r+=l;
									    ptmp->g+=l;
									    ptmp->b+=l;
									    }

									if(ptmp->r<los) ptmp->r=los;
									if(ptmp->r>his) ptmp->r=his;
									if(ptmp->g<los) ptmp->g=los;
									if(ptmp->g>his) ptmp->g=his;
									if(ptmp->b<los) ptmp->b=los;
									if(ptmp->b>his) ptmp->b=his;

									l=MAXDIST;
									for(k=((vrams == 1)?2:4);k--;)
									{
									int r, g, b;
									m=100-pixb;
									r = tvg[ (pixb*tvd[prevp.r] + m*tvd[base[k].r] ) / 100 ];
									g = tvg[ (pixb*tvd[prevp.g] + m*tvd[base[k].g] ) / 100 ];
									b = tvg[ (pixb*tvd[prevp.b] + m*tvd[base[k].b] ) / 100 ];
									
									    if(grey || (vrams == 1))
										m=(g-ptmp->g)*(g-ptmp->g);
									    else
										m=(r-ptmp->r)*(r-ptmp->r)+(g-ptmp->g)*(g-ptmp->g)+(b-ptmp->b)*(b-ptmp->b);
										if(m<l)
										{
											l=m;
											p=k;
										}
									}
									
									k=100-pixb;
									if(!dither)
									{
									prevp.r = ptmp->r - tvg[ (pixb*tvd[prevp.r] + k*tvd[base[p].r] ) / 100 ];
									prevp.g = ptmp->g - tvg[ (pixb*tvd[prevp.g] + k*tvd[base[p].g] ) / 100 ];
									prevp.b = ptmp->b - tvg[ (pixb*tvd[prevp.b] + k*tvd[base[p].b] ) / 100 ];
									}
									else
									{
									prevp.r = base[p].r - tvg[ (pixb*tvd[prevp.r] + k*tvd[base[p].r] ) / 100 ];
									prevp.g = base[p].g - tvg[ (pixb*tvd[prevp.g] + k*tvd[base[p].g] ) / 100 ];
									prevp.b = base[p].b - tvg[ (pixb*tvd[prevp.b] + k*tvd[base[p].b] ) / 100 ];
									}

									if((i&1) && (!dither))
									{
										atr[i/ay][(vrx-1-j)/PA_X].b[0][i%ay]=((p&2)<<6)+(atr[i/ay][(vrx-1-j)/PA_X].b[0][i%ay]>>1);
										atr[i/ay][(vrx-1-j)/PA_X].b[1][i%ay]=((p&1)<<7)+(atr[i/ay][(vrx-1-j)/PA_X].b[1][i%ay]>>1);
										o=prevp.r;
										(ptmp-1)->r+=7*o/16;
										(ptmp+VR_X+3)->r+=3*o/16;
										(ptmp+VR_X+2)->r+=5*o/16;
										(ptmp+VR_X+1)->r+=o/16;
										o=prevp.g;
										(ptmp-1)->g+=7*o/16;
										(ptmp+VR_X+3)->g+=3*o/16;
										(ptmp+VR_X+2)->g+=5*o/16;
										(ptmp+VR_X+1)->g+=o/16;
										o=prevp.b;
										(ptmp-1)->b+=7*o/16;
										(ptmp+VR_X+3)->b+=3*o/16;
										(ptmp+VR_X+2)->b+=5*o/16;
										(ptmp+VR_X+1)->b+=o/16;
										vtmp-=bpp;
										conv(vtmp,tvd[base[p].r],tvd[base[p].g],tvd[base[p].b] );
										ptmp--;
									}
									else
									{
										atr[i/ay][j/PA_X].b[0][i%ay]=((p&2)>>1)+(atr[i/ay][j/PA_X].b[0][i%ay]<<1);
										atr[i/ay][j/PA_X].b[1][i%ay]=(p&1)+(atr[i/ay][j/PA_X].b[1][i%ay]<<1);
										//if(!dither)
										//    {
										o=prevp.r;
										(ptmp+1)->r+=7*o/16;
										(ptmp+VR_X+1)->r+=3*o/16;
										(ptmp+VR_X+2)->r+=5*o/16;
										(ptmp+VR_X+3)->r+=o/16;
										o=prevp.g;
										(ptmp+1)->g+=7*o/16;
										(ptmp+VR_X+1)->g+=3*o/16;
										(ptmp+VR_X+2)->g+=5*o/16;
										(ptmp+VR_X+3)->g+=o/16;
										o=prevp.b;
										(ptmp+1)->b+=7*o/16;
										(ptmp+VR_X+1)->b+=3*o/16;
										(ptmp+VR_X+2)->b+=5*o/16;
										(ptmp+VR_X+3)->b+=o/16;
										//    }
										conv(vtmp,tvd[base[p].r],tvd[base[p].g],tvd[base[p].b]);
										vtmp+=bpp;
										ptmp++;
									}
									
								prevp=base[p];
								
								}
								if(dither)
								    ptmp+=VR_X+1-vrx;
								else
								{
								ptmp+=VR_X+2;
								vtmp+=vrx*bpp;
								}
							}
							printf("dithered, ");
							fflush(stdout);
							
							if(!col && (vrams == 2))
							  {
							ban("VRAM reordering...");
							XFlush(dis); 
							for(i=vry/ay;i--;)
							for(j=vrx/PA_X;j--;)
							{
								
								for(q=2;q--;)
								  {
								    r=s=0;
								  for(k=ay;k--;)
								    for(l=128;l;l>>=1)
								      if(atr[i][j].b[q][k]&l) s++; else r++;
								  if(s>r)
								    {
								      l=atr[i][j].a[q];
								    if(vic)
								      l=((HR_L&l)<<4) + ((l>>4)&HR_L);
								    else
								      l=((ZX_L&l)<<3) + ((l>>3)&ZX_L) + (l&BRIGHT);
								    atr[i][j].a[q]=l;
								      for(k=ay;k--;)
									atr[i][j].b[q][k]=~atr[i][j].b[q][k];
								    }
								  }
								r=s=0;
								decodeatr(i,j);
								m=RI*pure[1].r+GI*pure[1].g+BI*pure[1].b;
								n=RI*pure[0].r+GI*pure[0].g+BI*pure[0].b;
								o=RI*pure[3].r+GI*pure[3].g+BI*pure[3].b;
								p=RI*pure[2].r+GI*pure[2].g+BI*pure[2].b;
								for(k=ay;k--;)
								for(l=128;l;l>>=1)
								{
									r+=atr[i][j].b[0][k]&l?m:n;
									s+=atr[i][j].b[1][k]&l?o:p;
								}
								if(r>s)
								{
									q=atr[i][j].a[0];
									atr[i][j].a[0]=atr[i][j].a[1];
									atr[i][j].a[1]=q;
									for(k=ay;k--;)
									{
										q=atr[i][j].b[0][k];
										atr[i][j].b[0][k]=atr[i][j].b[1][k];
										atr[i][j].b[1][k]=q;
									}
								}
							}
							printf("ordered, ");
							  }
							ban("Conversion finished");
							XPutImage(dis,win,gc,xim,0,0,0,0,vrx,vry);
							fflush(stdout);
							XFlush(dis);
							m=0; l=2; n=1;
						}
				}
			}
			killwindow();
			for(m=2;m-- && ((m == 1) || (vrams == 2));)
			{
			if(vic)
			  {
			  for(j=0;j<vry/PA_Y;j++)
			  for(i=0;i<vrx/PA_X;i++)
			  for(l=0;l<PA_Y/ay;l++)
			  for(k=0;k<ay;k++)
			    fputc(atr[(PA_Y/ay)*j+l][i].b[m][k],tap);
			  }
			else
			  {
				c=0xff;
				fwrite(tapprefix,sizeof(tapprefix),1,tap);
				for(l=0;l<3;l++)
				for(n=0;n<PA_Y/ay;n++)
				for(k=0;k<ay;k++)
				for(j=0;j<PA_Y;j++)
				for(i=0;i<vrx/PA_X;i++)
				{
					d=atr[(vry/(3*ay))*l+(PA_Y/ay)*j+n][i].b[m][k];
					fputc(d,tap);
					c=c^d;
				}
			  }
				for(j=0;j<vry/ay;j++)
				for(i=0;i<vrx/PA_X;i++)
				{
					d=atr[j][i].a[m];
					fputc(d,tap);
					c^=d;
				}
				if(!vic) fputc(c,tap);
			}
			printf("saved.\n");
		}
		else
			printf("Not enough memory for viewing\n");
		if(!rgbtest)
		{
		if(format == F_JPEG)
		{
		jpeg_finish_decompress(&jdecs);
		jpeg_destroy_decompress(&jdecs);
		}
		fclose(file);
		}
	}
	else
		printf("\nCannot open image: %s\n",argv[actarg]);
	actarg++;
}

fclose(tap);

free(matrixes[7].m);

return OK;
}
