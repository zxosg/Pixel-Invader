ver		=	"0.42"
reds		=	newArray(256);
greens	=	newArray(256);
blues		=	newArray(256);
var pal1	=	newArray(128);
var pal2	=	newArray(128);
var pal3	=	newArray(128);
var ZXRed	=	newArray(8);
var ZXGrn	=	newArray(8);
var ZXBlu	=	newArray(8);
var wdth	=	getWidth();
var hght	=	getHeight();

var imageName	=	getTitle();
var imageID0	=	getImageID();
var imageID1	=	0;
var imageID2	=	0;
var imageID3	=	0;
var imageID4	=	0;
var ch		=	0;
var cl		=	0;
var p			= 	0;
var L			= 	newArray(0,2,4,7);
var rgb		= 	0;
var blk		= 	1;
var rgbv, rgbi, rlo, rhi;

	getLut(reds, greens, blues);
	// print(reds.length);

	if(reds.length==64)
		mode=0;
	else if(reds.length==16)
		mode=1;
	else if(reds.length==128)
		mode=2;
	else print("error, wrong palette");

	init(mode);

	setBatchMode(true);
	for(x=0;x<wdth;x++) {
		for(y=0;y<hght;y++) {
			selectImage(imageID0); col = getPixel(x,y);	
			if(mode==2) {
//				if(col == getPixel(x+1,y) && col == getPixel(x,y+1) && col == getPixel(x+1,y+1)) {
				if(col == getPixel(x+1,y)) {
					selectImage(imageID1); setPixel(x,y,pal1[col]);
					selectImage(imageID2);
					if(y%2) {
						setPixel(x<<1,y,pal2[col]);setPixel(1+x<<1,y,pal3[col]);
					} else {
						setPixel(x<<1,y,pal3[col]);setPixel(1+x<<1,y,pal2[col]);
					}
				} else {
					selectImage(imageID1); setPixel(x,y,pal1[col]);
					selectImage(imageID2); setPixel(x<<1,y,pal2[col]);setPixel(1+x<<1,y,pal3[col]);
				}
					
			} else if((x+y)&1) {
				selectImage(imageID1); setPixel(x,y,pal1[col]);
				selectImage(imageID2); setPixel(x,y,pal2[col]);
			} else {
				selectImage(imageID1); setPixel(x,y,pal2[col]);
				selectImage(imageID2); setPixel(x,y,pal1[col]);
			}

		}
	}

	setBatchMode(false);

	selectImage(imageID1); updateDisplay();
		run("Duplicate...", "title=temp1"); t1=getImageID(); run("RGB Color");
		if(mode==2) run("Size...", "width="+wdth*2+" height="+hght);
	selectImage(imageID2); updateDisplay();
		run("Duplicate...", "title=temp2"); t2=getImageID(); run("RGB Color");
		imageCalculator("Average create", "temp1","temp2");
		updateDisplay();
		selectImage(t1); close();
		selectImage(t2); close();

	if(mode==0) {
		imageID3 = createQLLores(imageID1,"LR1");
		imageID4 = createQLLores(imageID2,"LR2");
	} else if(mode==1) {
		imageID3 = createQLHires(imageID1,"HR1");
		imageID4 = createQLHires(imageID2,"HR2");
	} else {
		imageID3 = createQLLores(imageID1,"LR1");
		imageID4 = createQLHires(imageID2,"HR2");
	}
	selectImage(imageID3); rename(imageName+getTitle()); updateDisplay();
	selectImage(imageID4); rename(imageName+getTitle()); updateDisplay();
// end program

function createQLLores(img,name) {
	newImage(name, "8-bit White", 128, hght, 1); id = getImageID();
	setBatchMode(true);
	for(x=0;x<wdth;x+=4) {
		for(y=0;y<hght;y++) {
			selectImage(img);
			loByte = 0 ; hiByte = 0;
			p = getPixel(x  ,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); b = (ZXBlu[p]>0); loByte += (g<<7); hiByte += (r<<7 + b<<6); // print("0_p:"+p+" r:"+r+" g:"+g+" b:"+b+" lo:"+loByte+" hi:"+hiByte);
			p = getPixel(x+1,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); b = (ZXBlu[p]>0); loByte += (g<<5); hiByte += (r<<5 + b<<4); // print("1_p:"+p+" r:"+r+" g:"+g+" b:"+b+" lo:"+loByte+" hi:"+hiByte);
			p = getPixel(x+2,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); b = (ZXBlu[p]>0); loByte += (g<<3); hiByte += (r<<3 + b<<2); // print("2_p:"+p+" r:"+r+" g:"+g+" b:"+b+" lo:"+loByte+" hi:"+hiByte);
			p = getPixel(x+3,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); b = (ZXBlu[p]>0); loByte += (g<<1); hiByte += (r<<1 + b   ); // print("3_p:"+p+" r:"+r+" g:"+g+" b:"+b+" lo:"+loByte+" hi:"+hiByte);
			selectImage(id); setPixel(x>>1,y,loByte); setPixel(x>>1+1,y,hiByte);
		}
	}
	setBatchMode(false);
	updateDisplay();
	return id;
}

function createQLHires(img,name) {
	newImage(name, "8-bit White", 128, hght, 1); id = getImageID();
	setBatchMode(true);
	for(x=0;x<wdth<<1;x+=8) {
		for(y=0;y<hght;y++) {
			selectImage(img);
			loByte = 0 ; hiByte = 0;
			p = getPixel(x  ,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); loByte += (g<<7); hiByte += (r<<7); 
			p = getPixel(x+1,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); loByte += (g<<6); hiByte += (r<<6); 
			p = getPixel(x+2,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); loByte += (g<<5); hiByte += (r<<5); 
			p = getPixel(x+3,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); loByte += (g<<4); hiByte += (r<<4); 
			p = getPixel(x+4,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); loByte += (g<<3); hiByte += (r<<3); 
			p = getPixel(x+5,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); loByte += (g<<2); hiByte += (r<<2);
 			p = getPixel(x+6,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); loByte += (g<<1); hiByte += (r<<1); 
			p = getPixel(x+7,y); r = (ZXRed[p]>0); g = (ZXGrn[p]>0); loByte += (g   ); hiByte += (r   ); 
			selectImage(id); setPixel(x>>2,y,loByte); setPixel(x>>2+1,y,hiByte);
		}
	}
	setBatchMode(false);
	updateDisplay();
	return id;
}

function init(mode) {
	Dialog.create("QL Dithvide")
		Dialog.addCheckbox("RGB is 0xFF", rgb);
		Dialog.addNumber("RBG Lo (*)", 192);
		Dialog.addNumber("RBG Hi (*)", 255);
		Dialog.addCheckbox("Force Black", blk);
	Dialog.show();
	rgb = Dialog.getCheckbox();
	rlo = Dialog.getNumber();
	rhi = Dialog.getNumber();
	blk = Dialog.getCheckbox();

	if(rlo>rhi) rlo = rhi;

	if(rgb) {
		blk = 1;
		rlo = 255;
		rhi = 255;
	}

	rgbi = (rhi - rlo)/(7+blk);
	if(blk) {
		ZXRed[0]=0 ; ZXGrn[0]=0 ; ZXBlu[0]=0;
		rgbv = rlo;
		blak = 0;
	} else {
		ZXRed[0]=rlo ; ZXGrn[0]=rlo ; ZXBlu[0]=rlo;
		rgbv = rlo + rgbi;
		blak = rlo;
	}
	
	ZXRed[1]=blak ; ZXGrn[1]=blak ; ZXBlu[1]=rgbv; rgbv += rgbi;
	ZXRed[2]=rgbv ; ZXGrn[2]=blak ; ZXBlu[2]=blak; rgbv += rgbi;
	ZXRed[3]=rgbv ; ZXGrn[3]=blak ; ZXBlu[3]=rgbv; rgbv += rgbi;
	ZXRed[4]=blak ; ZXGrn[4]=rgbv ; ZXBlu[4]=blak; rgbv += rgbi;
	ZXRed[5]=blak ; ZXGrn[5]=rgbv ; ZXBlu[5]=rgbv; rgbv += rgbi;
	ZXRed[6]=rgbv ; ZXGrn[6]=rgbv ; ZXBlu[6]=blak; rgbv += rgbi;
	ZXRed[7]=rgbv ; ZXGrn[7]=rgbv ; ZXBlu[7]=rgbv; rgbv += rgbi;

	if(mode==0) {				// 0=low res. mode
		// print("low res mode");
		newImage(imageName+"_1", "8-bit White", wdth, hght, 1); setLut(ZXRed, ZXGrn, ZXBlu); imageID1	= getImageID();
		newImage(imageName+"_2", "8-bit White", wdth, hght, 1); setLut(ZXRed, ZXGrn, ZXBlu); imageID2	= getImageID();
		p = 0;
		for(i=0;i<8;i++) {
			for(j=0;j<8;j++) {
				pal1[p]=i; pal2[p]=j; p++;
			}
		}
	} else if(mode==1) {			// 1=hires mode
		// print("high res mode");
		newImage(imageName+"_1", "8-bit White", wdth, hght, 1); setLut(ZXRed, ZXGrn, ZXBlu); imageID1	= getImageID();
		newImage(imageName+"_2", "8-bit White", wdth, hght, 1); setLut(ZXRed, ZXGrn, ZXBlu); imageID2	= getImageID();
		for(i=0;i<4;i++) {
			for(j=0;j<4;j++) {
				pal1[p]=L[i] ; pal2[p]=L[j]; p++;
			}
		} 
	} else if(mode==2) {			// 2=mixed mode
		// print("mixed res mode");
		newImage(imageName+"_1", "8-bit White", wdth, hght, 1); setLut(ZXRed, ZXGrn, ZXBlu); imageID1	= getImageID();
		newImage(imageName+"_2", "8-bit White", wdth<<1, hght, 1); setLut(ZXRed, ZXGrn, ZXBlu); imageID2	= getImageID();
		for(i=0;i<8;i++) {
			for(j=0;j<4;j++) {
				for(k=0;k<4;k++) {
					pal1[p]=i ; pal2[p]=L[j] ; pal3[p]=L[k]; p++;
				}
			}
		}
	} else
		print("error, wrong mode");
}
