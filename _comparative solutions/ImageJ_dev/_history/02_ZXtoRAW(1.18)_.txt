function convert() {
	if (0==attrs) {
		selectImage(imageID0);
		makeRectangle(0, 0, getWidth(), getHeight());

		getStatistics(area, mean, pap, ink, std, attrCount);
		makeRectangle(sx, sy, wdth, hght);
	}

	bpixels = "";
	ctr = 0;

	for(y=sy; y<ey; y+=stepY) {
		for(x=sx;x<ex; x+=8) {
			selectImage(imageID0);
			if (attrs) {
				makeRectangle(x, y, 8, stepY);
				getStatistics(area, mean, pap, ink, std, attrCount);
				if (std==0) {
					ink = -1;
					pap = ink;
				}
				if (std==0 && chTrans==1)
						ink = 8;
					
				// if (std==0) { // = single color in attribute
				//	ink	= 8;
				// }
			}
			for(j=0;j<stepY;j++) {
				byte = 0;
				selectImage(imageID0);
				for(i=0;i<8;i++) {
					col = getPixel(x+i,y+j);
					byte = byte<<1+(col==ink);
					if(showPixels) {
						selectImage(imageID3);
						setPixel(x+i,y+j,(col==ink)*255);
						selectImage(imageID0);
					}
				}
				selectImage(imageID1);
				setPixel(x>>3,(y-sy)+j,byte);
				bpixels = bpixels + fromCharCode(byte);
			}

			bchar[ctr]=bpixels;
			bpixels = "";
			ctr++;

			if (attrs) {
				if (chTrans==1 && ink==8 && pap==nmTrans) {
					setPixel(x>>3,(y-sy)/stepY+hght,nmReplc);
				} else {
					setPixel(x>>3,(y-sy)/stepY+hght,pap<<3+ink-8*(ink==8));
				}
			}
		}
	}
}

function makeHDI() {
	newImage(imageName+"_hd", "8-bit White", 1, (3 + bwdth * (hght+attrs*(hght/stepY))), 1);
	imageID2 = getImageID();

	bhght = hght + attrs*(hght/stepY);
	addr	= 0;
	setPixel(0,addr,bwdth); addr++;
	setPixel(0,addr, hght); addr++;
	setPixel(0,addr,attrs*stepY); addr++;

  if (interleave) {
  	for(y=0; y<hght; y++) {
  		if (attrs && (y%stepY==0)) {
  			for(x=0; x<bwdth; x++) {
  				selectImage(imageID1);
  				val=getPixel(x,y/stepY+hght);
  				selectImage(imageID2);
  				setPixel(0,addr,val);
  				addr++;
  			}
  		}

  		for(x=0; x<bwdth; x++) {
  			selectImage(imageID1);
  			val=getPixel(x,y);
  			selectImage(imageID2);
  			setPixel(0,addr,val);
  			addr++;
  		}
  	}
  } else {
  	for(y=0; y<bhght; y++) {
  		for(x=0; x<bwdth; x++) {
  			selectImage(imageID1);
  			val=getPixel(x,y);
  			selectImage(imageID2);
  			setPixel(0,addr,val);
  			addr++;
  		}
  	}
  }
}

function init() {
	Dialog.create("8 col Indexed to ZX Raw")
		Dialog.addNumber("Attribute height", stepY)
		Dialog.addCheckbox("Interleave Attrs with Pixels",interleave)
		Dialog.addCheckbox("Include attributes", attrs)
		Dialog.addCheckbox("Align selection to attributes", align)
		Dialog.addCheckbox("Clear selection", clear)
		Dialog.addCheckbox("Create header",header)
		Dialog.addCheckbox("Show pixels",showPixels)
		Dialog.addCheckbox("Replace transparent color",chTrans)
		Dialog.addNumber("Transparent color", nmTrans)
		Dialog.addNumber("Replacement color", nmReplc)

	Dialog.show();
	stepY = Dialog.getNumber();
	interleave = attrs = Dialog.getCheckbox();
	attrs = Dialog.getCheckbox();
	align = Dialog.getCheckbox();
	clear = Dialog.getCheckbox();
	header = Dialog.getCheckbox();
	showPixels = Dialog.getCheckbox();
	chTrans = Dialog.getCheckbox();
	nmTrans = Dialog.getNumber();
	nmReplc = Dialog.getNumber();

	stepY = (attrs==0)+(attrs==1)*stepY;
	alignXY();

	if (clear) {
		run("Select None");
	}

	getSelectionBounds(sx, sy, wdth, hght);
	ex = sx + wdth;
	ey = sy + hght;
	bwdth = floor(wdth/8);
	attrPos = attrs*hght*bwdth;

	newImage(imageName+"_zx", "8-bit White", bwdth, hght+attrs*(hght/stepY), 1);
	imageID1 = getImageID();

	if (showPixels) {
		newImage(imageName+"_bw", "8-bit White", wdth, hght, 1);
		imageID3 = getImageID();
	}
}

function alignXY() {
	getSelectionBounds(x, y, w, h);
	x = round(x/8)*8;
	w = round(w/stepY)*stepY;
	if (attrs) {
		y = round(y/stepY)*stepY;
		h = round(h/stepY)*stepY;
	}
	makeRectangle(x,y,w,h);
}

function mx( bch ) { 			// mirror X of whole attr
	out="";
	for (i=0; i<8; i++) {
		b = charCodeAt( bch, i );	
		b = (b & 0xF0) >> 4 | ((b & 0x0F) << 4);
		b = (b & 0xCC) >> 2 | ((b & 0x33) << 2);
		b = (b & 0xAA) >> 1 | ((b & 0x55) << 1);
		out=out+fromCharCode(b);
	}
	return out;
}

function my( bch ) {			// mirror y of whole attr
	out="";
	for (i=7; i>=0; i--) {
		out = out + substring( bch, i,i+1);
	}
	return out;
}

function oldrr( bch ) {			// rotate right whole attr
	out="";
	for (j=7; j>=0 ; j--) {
		b=0;
		for (i=0; i<8; i++) {
			b=b | (((charCodeAt( bch, i) >> j) & 1) << i);
		}
		out = out + fromCharCode(b);
	}
	return out;
}

function rr( bch ) {			// rotate right whole attr
	out="";
	for (j=7; j>=0 ; j--) {
		b=0;
		for (i=0; i<8; i++) {
			b=b<<1 + ((charCodeAt( bch, i) >> j) & 1);
		}
		out = out + fromCharCode(b);
	}
	return out;
}

function ff( bch ) {			// invert arttr
	out="";
	for (i=0; i<8; i++) {
		out = out + fromCharCode(charCodeAt( bch, i ) ^ 0xff) ; 
	}
	return out;
}

function charTransform() {

	for (k=0; k<16; k++) {
		print ("set no:"+k);
		for (i=0; i<blen; i++) {
			b=bchar[i];		// destination
//			print(i+" "+b.length);
			if (b=="")
				i=768;
			if (k&1==1)
				b=mx(b);
			if (k&2==2)
				b=my(b);
			if (k&4==4)
				b=rr(b);
			if (k&8==8)
				b=ff(b);
			btran[i+(768*k)]=b;
		}
	}
}

function charCleanUp() {
	Array.sort(bchar);
	j=0;
	for (i=1; i<bchar.length; i++) {
		bi=bchar[i];
		bj=bchar[j];
		if (bchar[j]!=bchar[i]) {
			j++;
			bchar[j]=bchar[i];
			bchar[i]="";
		}
	}
	blen=j;
			
}

function charStats() {
	Array.fill(bmap,-1);
	Array.fill(bocur,0);
	for (i =0; i<blen; i++) {
		for (j=0; j<blen; j++) {
			a=bchar[i];		// source
			for (k=0; k<16; k++) {
				b=btran[j+(768*k)];		// destination

				if (b==a) {
					bocur[i]=bocur[i]+1;
					bmap[i]=j;
					j=blen;
					k=16;
				}

//				if ((b==a) && (bmap[i]==-1)) {
//					bocur[i]=bocur[i]+1;
//					bmap[i]=j;
//					k=16;
//					j=blen;
//				}
			}
		}
	}
}

function mapCleanUp2() {
	Array.fill(bocur,0);
	for (i=0; i<768; i++) {
		pos=bmap[i];
		bocur[pos]=bocur[pos]+1;
	}
	Array.sort(bocur);
	Array.reverse(bocur);
}

function mapCleanUp() {

	Array.sort(bmap);
	Array.reverse(bmap);

	j=0;
	i=1;
	do {
		if(bmap[j]!=bmap[i]) {
			j++;
			bmap[j]=bmap[i];
		}
		i++;
	} while (i<768);
	blen=j+1;
}


function mapCleanUp_old() {

	Array.sort(bmap);
	Array.reverse(bmap);

	j=0;
	i=1;
	do {
		if(bmap[j]!=bmap[i]) {
			j++;
			bmap[j]=bmap[i];
		}
		i++;
	} while (bmap[i]!=-1);
	blen=j+1;
}
