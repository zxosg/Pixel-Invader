var sID;
var dID;
var w;
var h;
var m = 4;
v = newArray(0,1,6,7);
o = newArray(0,4,3,7);
c1 = 0xff0000
c2 = 0x00ff00
c3 = 0x0000ff
c4 = 0x808080

// source image
sID = getImageID();
w = getWidth();
h = getHeight();

// destination image
newImage("image", "RGB black", w*m, h*m, 1);
dID = getImageID();

setBatchMode(true);
for(x=0;x<w;x++) {
	for(y=0;y<h;y++) {
		selectImage(sID); 
		col = getPixel(x,y);
		selectImage(dID);
		r = col>>22&3; r=v[r];
		g = col>>14&3; g=o[g];
		b = col>>06&3; b=v[b];
		i = (col>>16&255 + col>>8&255 + col&255);	i = i/3; i = i>>6; i=o[i];
		xx = x*m;
		yy = y*m;
		putPixel(xx,yy  ,r>>2&1*c1); putPixel(xx+1,yy  ,r>>1&1*c1); putPixel(xx+2,yy  ,r&1*c1);
		putPixel(xx,yy+1,g>>2&1*c2); putPixel(xx+1,yy+1,g>>1&1*c2); putPixel(xx+2,yy+1,g&1*c2);
		putPixel(xx,yy+2,b>>2&1*c3); putPixel(xx+1,yy+2,b>>1&1*c3); putPixel(xx+2,yy+2,b&1*c3);	
		putPixel(xx,yy+3,i>>2&1*c4); putPixel(xx+1,yy+3,i>>1&1*c4); putPixel(xx+2,yy+3,i&1*c4);
	}
}
