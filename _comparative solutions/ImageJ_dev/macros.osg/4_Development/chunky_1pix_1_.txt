var sID;
var dID;
var w;
var h;
var sx = 2;
var sy = 2;

// pixels for odd and even lines (dithering pattern)
if (sx<=2) {
	v = newArray(0,4,6,7,0,2,6,7);
} else {
//	v = newArray(0,1,3,7,0,4,6,7);
	v = newArray(0,3,7,7,0,6,6,7);

}
// order of rgb scanlines 
//           r g b i
l = newArray(0,1,2,3);

c1 = 0xff0000;
c2 = 0x00ff00;
c3 = 0x0000ff;
c4 = 0xc0c0c0;

// source image
sID = getImageID();
w = getWidth();
h = getHeight();

// destination image
newImage("image", "RGB black", w, h, 1);
dID = getImageID();

setBatchMode(true);
for(x=0;x<w;x+=sx) {
	for(y=0;y<h;y+=sy) {
		selectImage(sID); 
		col = getPixel(x,y);
		selectImage(dID);
		xx = x;
		yy = y;
		r = col>>22&3; r=v[r+l[0]&1*4];
		g = col>>14&3; g=v[g+l[1]&1*4];
		b = col>>06&3; b=v[b+l[2]&1*4];
		i = (col>>16&255 + col>>8&255 + col&255);	i = i/3; i = i>>6; i=v[i+l[3]&1*4];
		putPixel(xx,yy+l[0],r>>2&1*c1); putPixel(xx+1,yy+l[0],r>>1&1*c1); putPixel(xx+2,yy+l[0],r&1*c1);
		putPixel(xx,yy+l[1],g>>2&1*c2); putPixel(xx+1,yy+l[1],g>>1&1*c2); putPixel(xx+2,yy+l[1],g&1*c2);
		putPixel(xx,yy+l[2],b>>2&1*c3); putPixel(xx+1,yy+l[2],b>>1&1*c3); putPixel(xx+2,yy+l[2],b&1*c3);	
		putPixel(xx,yy+l[3],i>>2&1*c4); putPixel(xx+1,yy+l[3],i>>1&1*c4); putPixel(xx+2,yy+l[3],i&1*c4);
	}
}
