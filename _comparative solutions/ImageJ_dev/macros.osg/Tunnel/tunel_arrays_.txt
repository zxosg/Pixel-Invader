// http://lodev.org/cgtutor/tunnel.html

var w=256;
var h=128;
var texHeight=256;
var texWidth =256;
var ratio = 32;
var angle;
var distance;
var distanceTable   = newArray(w*h);
var angleTable      = newArray(w*h);
var texture         = newArray(texWidth*texHeight);
var dID, aID, nID, tID;
var ds=distanceTable.length;
var as=angleTable.length;
var ts=texture.length;
var texX, texY;

// newImage("texture", "8-bit black", texWidth, texHeight, 1); tID = getImageID();

// generate texture

    for(x = 0; x < texWidth; x++) {
    for(y = 0; y < texHeight; y++) {
        texture[x+y*texWidth] = (x * 256 / texWidth) ^ (y * 256 / texHeight);
//        setPixel(x,y,texture[x+y*texWidth]);
    }
    }

// generate mapping tables

    setBatchMode(true);
    for(x = 0; x < w; x++) {
    for(y = 0; y < h; y++) {

		distance = round(ratio * texHeight / sqrt((x - w/2.0) * (x - w/2.0) + (y - h/2.0) * (y - h/2.0))) % texHeight;
		angle = round(0.5 * texWidth * atan2(y - h/2.0, x - w/2.0) / PI);
		// if (angle<0)
			// angle=256+angle;
			
        distanceTable[x+y*w] = distance;
        angleTable[x+y*w] = angle;

    }
    }
    setBatchMode(false);

newImage("tunnel", "8-bit black", w, h, 1); nID = getImageID();

// animate textures

    for(l = 0; l < 200; l++) {
    setBatchMode(true);
    animation = getTime() / 10000.0;

    //calculate the shift values out of the animation value
    shiftX = floor(texWidth * 1.0 * animation);
    shiftY = floor(texHeight * 0.7 * animation);        
        
    for(x = 0; x < w; x++) {
    for(y = 0; y < h; y++) {
        
        texY = (distanceTable[x+y*w] + round(shiftY)) % texHeight;
        texX = (angleTable[x+y*w] + round(shiftX)) % texWidth;
        color = texture[texX+texY*texWidth];

        //get the texel from the texture by using the tables, shifted with the animation values
        selectImage(nID); setPixel(x,y,color);
    }
    }
    setBatchMode(false);
    updateDisplay();
    // wait(10);

}
