// dither

var hght	=	50; //getHeight();
var dither	= 	50;		// 0 - 100
var d100	= 	dither/100;
var info	=	newArray(10);
var Index;
var imageName	=	getTitle();
var id0	=	getImageID();
var id1;
var id2;
var md	=	255*255+255*255+255*255+1;

IniRGBPalette(0);
run("Duplicate...", "title=temp1"); id1=getImageID();
newImage(imageName+"_reduced", "8-bit White", wdth, hght, 1); id2 = getImageID();
setLut(Rpal,Gpal,Bpal);
setBatchMode(true);
for(y=0; y<hght; y++) {
	showProgress(y/hght);
	for(x=0; x<wdth; x++) {
		selectImage(id1);
		rgb0	= getPixel(x,y);
		r0=(rgb0>>16)&0xff; g0=(rgb0>>8)&0xff; b0= rgb0&0xff;
		minDst = md;
	
		for(i=0; i<Rpal.length; i++) {
			Rdiff = r0 - Rpal[i];
			Gdiff = g0 - Gpal[i];
			Bdiff = b0 - Bpal[i];
			dstSq = Rdiff*Rdiff + Gdiff*Gdiff + Bdiff*Bdiff;
			if (dstSq < minDst) {
				minDst = dstSq;
				Index = i;
			}
		}
		Rerr	= (r0 - Rpal[Index])>>4*d100;
		Gerr	= (g0 - Gpal[Index])>>4*d100;
		Berr	= (b0 - Bpal[Index])>>4*d100;

		rgb1 = getPixel(x+1,y  ); r1=(rgb0>>16)&0xff+Rerr*7; g1=(rgb0>>8)&0xff+Gerr*7; b1=rgb0&0xff+Berr*7;
		rgb2 = getPixel(x-1,y+1); r2=(rgb0>>16)&0xff+Rerr*3; g2=(rgb0>>8)&0xff+Gerr*3; b2=rgb0&0xff+Berr*3;
		rgb3 = getPixel(x  ,y+1); r3=(rgb0>>16)&0xff+Rerr*5; g3=(rgb0>>8)&0xff+Gerr*5; b3=rgb0&0xff+Berr*5;
		rgb4 = getPixel(x+1,y+1); r4=(rgb0>>16)&0xff+Rerr  ; g4=(rgb0>>8)&0xff+Gerr  ; b4=rgb0&0xff+Berr  ;
		
		setPixel(x+1,y  ,h(r1)<<16+h(g1)<<8+h(b1));
		setPixel(x-1,y+1,h(r2)<<16+h(g2)<<8+h(b2));
		setPixel(x  ,y+1,h(r3)<<16+h(g3)<<8+h(b3));
		setPixel(x+1,y+1,h(r4)<<16+h(g4)<<8+h(b4));
		selectImage(id2);
		setPixel(x,y,Index);
	}
}
setBatchMode(false);
