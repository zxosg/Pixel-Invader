// http://lodev.org/cgtutor/tunnel.html

var 	w=128;
var	h=128;
var	texHeight=255
var	texWidth =255
var	ratio = 32;
var	angle;
var	distance;
var	distanceTable	= newArray(w*h);
var	angleTable	= newArray(w*h);
var	texture	= newArray(texWidth*texHeight);
var	dID, aID, nID;

newImage("texture", "8-bit black", texWidth, texHeight, 1);

// generate texture

	for(x = 0; x < texWidth; x++) {
	for(y = 0; y < texHeight; y++) {
		texture[x*y] = (x * 256 / texWidth) ^ (y * 256 / texHeight);
		setPixel(x,y,texture[x*y]);
	}
	}

// print("textures done");

// generate mapping tables
newImage("distn", "8-bit black", w, h, 1); dID=getImageID();
newImage("angle", "8-bit black", w, h, 1); aID=getImageID();


	setBatchMode(true);
	for(x = 0; x < w; x++) {
	for(y = 0; y < h; y++) {

		distance = round(ratio * texHeight / sqrt((x - w/2.0) * (x - w/2.0) + (y - h/2.0) * (y - h/2.0))) % texHeight;
		angle = round(0.5 * texWidth * atan2(y - h/2.0, x - w/2.0) / PI);
		if (angle<0)
			angle=256+angle; 
		distanceTable[x*y] = distance;
		angleTable[x*y] = angle;
		// print(x+" "+y+" "+angle);
		selectImage(dID); setPixel(x,y,distance);
		selectImage(aID); setPixel(x,y,angle);
	}
	}
	setBatchMode(false);

// print("tables done");


newImage("tunnel", "8-bit black", w, h, 1); nID = getImageID();

title1 = "Color Table";
title2 = "["+title1+"]";
f = title2;

run("Table...", "name="+title2+" width=250 height=400");
print(f, "\\Headings:n\tColX\tColY");

// animate textures

	for(l = 0; l < 200; l++) {
	setBatchMode(true);
	animation = getTime() / 10000.0;

	//calculate the shift values out of the animation value
	shiftX = floor(texWidth * 1.0 * animation);
	shiftY = floor(texHeight * 0.7 * animation);        
        
	for(x = 0; x < w; x++) {
	for(y = 0; y < h; y++) {
		
		selectImage(dID); texX = (getPixel(x,y) + round(shiftX)) % texWidth;
       	selectImage(aID); texY = (getPixel(x,y) + round(shiftY)) % texHeight;
		// print(f, "\t"+texX+"\t"+texY);

		//get the texel from the texture by using the tables, shifted with the animation values
		selectImage(nID); setPixel(x,y,texture[texX*texY]);
	}
	}
	setBatchMode(false);
	updateDisplay();
	wait(10);

}
print("over");
